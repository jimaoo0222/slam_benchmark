快速开始

git clone https://github.com/jimaoo0222/slam_benchmark.git
cd slam_benchmark

# 1) 下载一个 bag 放进 rosbag/（见「数据集」页）
# 2) 跑一个算法（包名 01_/02_ 自动用 MID-360s 外参，03_/04_ 自动用 C01 外参）
bash scripts/05_run_fastlio_on_bag.sh rosbag/04_02 results/run/fastlio

# 3) 对卷尺真值评测
python3 scripts/14_eval_on_site.py results/run/fastlio/traj.txt \
    --site config/site_mid360_path1_open.json \
    --stop-from results/run/fastlio/traj.txt
